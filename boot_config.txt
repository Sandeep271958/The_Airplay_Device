# =====================================================================
# Add these lines to /boot/firmware/config.txt
# =====================================================================

# Disable the default onboard audio (we use the I2S DAC instead)
dtparam=audio=off

# Enable the I2S DAC overlay (PCM5102A is hifiberry-dac compatible)
dtoverlay=hifiberry-dac

# Enable SPI interface (used by WS2812 LED driver)
dtparam=spi=on

# Enable Software SPI (Bus 1) for the ST7789 TFT display
dtoverlay=spi-gpio,sck_pin=5,mosi_pin=6,miso_pin=13,cs0_pin=25,spi_bus=1
