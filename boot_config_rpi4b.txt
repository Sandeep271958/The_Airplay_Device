# =====================================================================
# RPi 4B PROTOTYPE — Add these lines to /boot/firmware/config.txt
# =====================================================================
# Use this config while prototyping on RPi 4B without the I2S DAC.
# Switch to boot_config.txt when moving to RPi Zero 2W + PCM5102A.
# =====================================================================

# KEEP onboard audio ENABLED (for 3.5mm headphone jack output)
dtparam=audio=on

# Do NOT load hifiberry-dac overlay (no I2S DAC connected yet)
# Uncomment the line below once you connect the PCM5102A:
# dtoverlay=hifiberry-dac

# Enable SPI interface (used by WS2812 LED driver on GPIO 10)
dtparam=spi=on

# Enable Software SPI (Bus 1) for the ST7789 TFT display
dtoverlay=spi-gpio,sck_pin=5,mosi_pin=6,miso_pin=13,cs0_pin=25,spi_bus=1

# Force audio output to 3.5mm jack (not HDMI)
hdmi_ignore_edid_audio=1
